     {['Mon','Tue','Wed','Thu','Fri','Sat','Sun'].map(d=><b key={d}>{d}</b>)}
     {[...Array(first)].map((_,i)=><i key={'b'+i}/>)}
     {[...Array(len)].map((_,i)=>{
      const d=`${y}-${pad(m+1)}-${pad(i+1)}`,x=st(id,d);
      return (

       </div>

     })}

    <p className="note">Status labels: Present (green) Â· Late (amber) Â· Absent (red) Â· On Leave (blue) Â· Weekend / Off (neutral). A Saturday with hours indicates a scheduled shift.</p>
   </section>
   <section>

    <div className="scroll">
     <table>
      <thead>
       <tr><th>Date</th><th>Check-in</th><th>Check-out</th><th>Total hours</th><th>Status</th></tr>
      </thead>
      <tbody>
       {recent.map(d=>{

        return (
         <tr key={d}>
          <td>{d}</td>

 );

function Guard({role}){
 const {user}=useApp();
 if(!user)return <Navigate to="/login" replace/>;
 if(user.role!==role)return <Navigate to={'/'+user.role} replace/>;
 return <Shell role={role} key={role}/>;
}

export default function App(){
 return (
  <Provider>
   <Routes>
    <Route path="/" element={<Landing/>}/>

    {['admin','hr','employee'].map(r=><Route key={r} path={'/'+r} element={<Guard role={r}/>}/>)}
    <Route path="*" element={<Navigate to="/" replace/>}/>
   </Routes>
  </Provider>
 );
}
